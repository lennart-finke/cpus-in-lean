# Known QEMU (TCG) deviations from the Intel SDM in the system tests. run.sh applies each rule to the
# QEMU transcript (Python re.sub on whole lines) before comparing with the model; every other line
# must be identical. Format: REGEX <TAB> REPLACEMENT <TAB> # reason
^(pae\.(rsvd50|xd\.nonxe|2m\.rsvd13): vec=0000000e err=0000000)8	\g<1>9	# reserved-bit #PF: P=1 (SDM 4.7: the fault is on a present entry; hardware reports 0x9). TCG reports P=0.
^pae\.cr3\.rsvd: vec=0000ffff$	pae.cr3.rsvd: vec=0000000d	# MOV CR3 with a reserved bit in a present PDPTE raises #GP(0) (SDM 4.4.1). TCG does not check PDPTEs on MOV CR3.
^pae\.cr3: 00000000$	pae.cr3: 00000001	# (same: CR3 is unchanged after the #GP)
^pae\.pdpte\.cached: vec=0000000e .*$	pae.pdpte.cached: ok	# PDPTEs live in registers loaded by MOV CR3 (SDM 4.4.1); TCG re-reads them from memory on every walk.
^pae\.efer\.rsvd: vec=0000ffff$	pae.efer.rsvd: vec=0000000d	# WRMSR setting a reserved EFER bit raises #GP(0) (SDM WRMSR); TCG ignores the bit.
^db\.two: n=1 \[dr6=ffff0ff1 rf=0\]$	db.two: n=1 [dr6=ffff0ff3 rf=0]	# every breakpoint whose condition matches sets its Bn in DR6 (SDM 17.2.3); TCG reports only the first watchpoint.
^db\.movss\.at_n2: 0$	db.movss.at_n2: 1	# MOV SS / POP SS hold debug exceptions until after the next instruction (SDM 6.8.3, the CVE-2018-8897 behaviour); TCG delivers the #DB right after MOV SS.
^db\.movss\.int\.at_handler: 0 (.*)$	db.movss.int.at_handler: 1 \1	# (same: after MOV SS; INT n the #DB is taken at the first instruction of the INT handler)
^db\.rep: n=1 \[dr6=ffff0ff8 rf=0\]$	db.rep: n=1 [dr6=ffff0ff8 rf=1]	# a data breakpoint in a REP string instruction traps after the matching iteration (SDM 17.3.1.2), mid-instruction with RF set; TCG traps after the whole instruction.
^db\.rep\.ecx: 00000000 at_insn=0$	db.rep.ecx: 00000004 at_insn=1	# (same: ECX holds the remaining count, EIP the string instruction)
^task\.commit: (vec=0000000a err=000000f0) at_g=0 (.*)$	task.commit: \1 at_g=1 \2	# a fault after the task-switch commit point is raised in the new task before its first instruction (SDM 7.3), so the pushed EIP is the new task's; TCG pushes the old task's JMP and then runs off the rails.
^apic\.swdisable\.write: lvt_timer=00000066$	apic.swdisable.write: lvt_timer=00010066	# a software-disabled local APIC keeps every LVT mask bit set; "attempts to reset these bits will be ignored" (SDM 10.4.7.2). TCG lets them clear.
^apic\.reenable: n=1 \[00000067 .*$	apic.reenable: n=0	# a software-disabled local APIC only responds to INIT/NMI/SMI/SIPI (SDM 10.4.7.2), so the fixed self-IPI is dropped; TCG queues it in IRR and delivers it after re-enabling.
^(smm\.[a-z]+\.more: cs_sel=00000008 cs_attr=0000c09)a (.*)$	\1b \2	# loading CS by a far JMP sets the accessed bit of its descriptor (SDM 3.4.5.1), and the descriptor cache saved in SMRAM holds it; TCG never sets the accessed bit on protected-mode far jumps.
^smm\.gdt\.cs\.accessed: 0$	smm.gdt.cs.accessed: 1	# (same: the GDT entry itself)

^(x87 \d+ op=8 cw=[0-9a-f]+ a 0000[08]000_(?!0{16})[0-9a-f]{16} b (?!0000[7f]fff_(?!8000000000000000)[89a-f])(?:0000[08]000_[0-9a-f]{16}|0000[0-9a-f]{4}_[89a-f][0-9a-f]{15}) .* de=)0$	\g<1>1	# FSCALE reports a denormal operand (#D in its SDM exception list) when the result uses the operands; TCG misses it unless a NaN is involved.
